We summarize the finalized CRIME-Q appraisal for the included studies. We use the adapted items defined in the appraisal workbook to assess quality of reporting, methodological quality, and risk of bias (Andersen et al. 2024). The code below reads the item definitions and ratings from the same workbook so their order remains aligned.

NoteVariable definitions
Variable Definition
study_id Unique identifier for each included study.
QoR Quality of reporting.
MQ Methodological quality.
RoB Risk of bias.
Rating Recorded response for an appraisal item: Yes, Partly, No, Unclear, or NA, where applicable.

We load the finalized study ratings and adapted item definitions from the CRIME-Q workbook.

Show data-loading code 
# Read the finalized study-level ratings.
appraisal <- readxl::read_excel(
  path = input_file,
  sheet = "CRIME-Q_Appraisal",
  col_types = "text",
  na = ""
) |>
  dplyr::mutate(
    dplyr::across(
      dplyr::everything(),
      ~ dplyr::na_if(stringr::str_squish(.x), "")
    )
  )

# Read the codebook table below the rating legend.
codebook_raw <- readxl::read_excel(
  path = input_file,
  sheet = "Codebook",
  skip = 7,
  col_types = "text",
  na = ""
)

# Assign concise names to the five codebook columns.
names(codebook_raw)[1:5] <- c(
  "item",
  "item_name",
  "domain",
  "question",
  "criteria"
)

# Match the rating options recorded at the end of each question.
rating_pattern <- "(Yes|Partly|No|Unclear|NA)(/(Yes|Partly|No|Unclear|NA))+$"

# Keep the 21 scored items and omit their paired notes fields.
codebook <- codebook_raw |>
  dplyr::mutate(
    dplyr::across(
      dplyr::everything(),
      ~ dplyr::na_if(stringr::str_squish(.x), "")
    )
  ) |>
  dplyr::filter(
    domain %in% c("QoR", "MQ", "RoB"),
    !stringr::str_detect(item, "n$|_n$")
  ) |>
  dplyr::mutate(
    item_key = stringr::str_remove_all(item, "[^[:alnum:]]"),
    ratings = stringr::str_replace_all(
      stringr::str_extract(question, rating_pattern),
      "/",
      "; "
    ),
    definition = stringr::str_squish(
      stringr::str_remove(
        question,
        paste0("\\s*", rating_pattern)
      )
    )
  )
Show validation code 
# Select rating columns while excluding notes and appraisal time.
score_columns <- names(appraisal)[
  stringr::str_detect(names(appraisal), "^[0-9]+[A-Z]") &
    !stringr::str_detect(names(appraisal), "notes$")
]

# Match workbook column prefixes to the item codes in the codebook.
score_keys <- stringr::str_extract(score_columns, "^[^_]+")

if (!identical(score_keys, codebook$item_key)) {
  stop("The CRIME-Q rating columns do not match the codebook item order.")
}

if (anyDuplicated(appraisal$study_id) > 0L || anyNA(appraisal$study_id)) {
  stop("Each CRIME-Q row must have one unique study_id.")
}

# Convert the rating matrix to long format for validation and plotting.
item_map <- tibble::tibble(
  score_column = score_columns,
  item = codebook$item,
  domain = codebook$domain
)

plot_data <- appraisal |>
  dplyr::select(study_id, dplyr::all_of(score_columns)) |>
  tidyr::pivot_longer(
    cols = dplyr::all_of(score_columns),
    names_to = "score_column",
    values_to = "rating"
  ) |>
  dplyr::left_join(item_map, by = "score_column")

valid_ratings <- c("Yes", "Partly", "No", "Unclear", "NA")

if (anyNA(plot_data$rating)) {
  stop("The CRIME-Q appraisal contains a missing rating.")
}

unexpected_ratings <- setdiff(unique(plot_data$rating), valid_ratings)

if (length(unexpected_ratings) > 0L) {
  stop(
    "Unexpected CRIME-Q rating: ",
    paste(unexpected_ratings, collapse = ", ")
  )
}

if (anyNA(codebook$ratings)) {
  stop("A CRIME-Q item does not contain a recorded rating list.")
}

Study-level ratings

The workbook contains 16 studies and 21 scored CRIME-Q items. We plot the finalized ratings below. Notes and appraisal time remain in the workbook but do not enter the figure.

Show figure code 
# Create a concise study label from the study identifier.
study_labels <- appraisal |>
  dplyr::transmute(
    study_id,
    study = paste(
      stringr::str_to_title(stringr::str_extract(study_id, "^[^_]+")),
      "et al.",
      stringr::str_extract(study_id, "(19|20)[0-9]{2}")
    )
  ) |>
  dplyr::arrange(study_id)

# Set the item and study order used in the heatmap.
plot_data <- plot_data |>
  dplyr::left_join(study_labels, by = "study_id") |>
  dplyr::mutate(
    item = factor(item, levels = codebook$item),
    study = factor(study, levels = rev(study_labels$study)),
    rating = factor(rating, levels = valid_ratings)
  )

# Use one colour for each recorded rating.
rating_colours <- c(
  "Yes" = "#009E60",
  "Partly" = "#EFBF04",
  "No" = "#DC143C",
  "Unclear" = "#F4B183",
  "NA" = "#D9D9D9"
)

# Plot one tile for every study-item rating.
ggplot2::ggplot(
  plot_data,
  ggplot2::aes(x = item, y = study, fill = rating)
) +
  ggplot2::geom_tile(color = "#FFFDF8", linewidth = 0.65) +
  ggplot2::scale_x_discrete(position = "top") +
  ggplot2::scale_fill_manual(
    values = rating_colours,
    breaks = valid_ratings,
    drop = FALSE,
    name = NULL
  ) +
  ggplot2::labs(x = NULL, y = NULL) +
  ggplot2::theme_minimal(base_size = 12) +
  ggplot2::theme(
    panel.grid = ggplot2::element_blank(),
    axis.text.x = ggplot2::element_text(
      face = "bold",
      colour = "#242321",
      size = 10
    ),
    axis.text.y = ggplot2::element_text(
      face = "bold",
      colour = "#242321",
      size = 10
    ),
    axis.ticks = ggplot2::element_blank(),
    legend.position = "bottom",
    legend.text = ggplot2::element_text(size = 11),
    legend.key.width = grid::unit(1.5, "lines"),
    plot.margin = ggplot2::margin(10, 10, 10, 10)
  )
