Uni-moderators

  • Show All Code
  • Hide All Code
  • View Source

We fit one uni-moderator multilevel meta-regression for each study characteristic and effect-size type. The models use lnRR, lnVR, or lnCVR as the response and the same random-effects structure as the overall-effect model: study (study_id), effect size (es_id), and strain (strain). We construct a complete VCV matrix for each effect-size type with rho = 0.5. For categorical moderators, we remove levels with fewer than five effect sizes (\(k < 5\)) and subset the corresponding rows and columns of the matching VCV matrix.

NoteVariable definitions
Variable Definition
\(k\) Number of effect sizes in a model.
lnRR Log response ratio, which measures differences in group means.
lnVR Log variability ratio, which measures absolute variability.
lnCVR Log coefficient of variation ratio, which measures variability relative to the group mean.
V Sampling variance-covariance matrix for one effect-size type.
\(\rho\) Working correlation for multiple or repeated outcomes that share a group.
QM Omnibus test statistic for the moderator.
Marginal \(R^2\) Proportion of model variance associated with the moderator.
Conditional \(R^2\) Proportion of model variance associated with the moderator and random effects.

Across every lnRR plot, negative values represent less anxiety- or depression-like behaviour and positive values represent more anxiety- or depression-like behaviour. For lnVR and lnCVR, negative values represent lower variability and positive values represent greater variability in the treatment group.

We load the database created in Effect size calculation and construct a complete VCV matrix for lnRR, lnVR, and lnCVR using the same group identifiers as Sampling variance-covariance matrices. The lnVR and lnCVR datasets exclude percentage outcomes because their effect-size values are missing by design.

construct_metric_data <- function(data, effect, variance, rho = 0.5) {
  # Retain only rows available for this effect-size type.
  idx <- which(is.finite(data[[effect]]) & is.finite(data[[variance]]))
  metric_data <- data[idx, , drop = FALSE]

  # Construct the matching complete VCV matrix.
  V <- metafor::vcalc(
    vi = metric_data[[variance]],
    cluster = metric_data$study_id,
    grp1 = metric_data$ex_id,
    grp2 = metric_data$con_id,
    w1 = metric_data$ex_n,
    w2 = metric_data$c_n,
    obs = metric_data$es_id,
    rho = rho
  )

  list(data = metric_data, V = V)
}

Match category colours across plots

We assign one named colour to each category in the complete database. The named colours keep a category visually consistent when the retained levels differ among lnRR, lnVR, and lnCVR.

Code 
# Retain the categorical moderators and their missing-value labels.
categorical_moderators <- moderators |>
  filter(kind == "categorical")

# Create one named colour palette for each categorical moderator.
category_palettes <- purrr::map2(
  categorical_moderators$variable,
  categorical_moderators$missing_label,
  function(variable, missing_label) {
    category_levels <- sort(unique(as.character(db[[variable]])))
    category_levels <- category_levels[!is.na(category_levels)]

    # Exclude a category that the analysis treats as missing.
    if (!is.na(missing_label)) {
      category_levels <- category_levels[category_levels != missing_label]
    }

    # orchaRd capitalizes the first letter of plotted moderator levels.
    # Name the palette with those displayed labels so ggplot can match every
    # category instead of falling back to grey. Because this palette is built
    # once per moderator, each level keeps its colour across all three metrics.
    plotted_levels <- category_levels
    substr(plotted_levels, 1, 1) <- toupper(substr(plotted_levels, 1, 1))

    stats::setNames(
      grDevices::hcl.colors(length(category_levels), palette = "Dark 3"),
      plotted_levels
    )
  }
)

Prepare the moderator data

For a categorical moderator, we remove levels with \(k < 5\) and subset the same rows and columns from V. For a continuous moderator, we retain finite values and subset V in the same way.

Code 
prepare_categorical <- function(data, V, moderator, missing_label = NA_character_, min_k = 5) {
  # Start with rows that contain a recorded moderator value.
  usable <- !is.na(data[[moderator]])

  # Treat a specified text label as missing when required.
  if (!is.na(missing_label)) {
    usable <- usable & as.character(data[[moderator]]) != missing_label
  }

  # Count the effect sizes in each moderator level.
  level_counts <- data[usable, ] |>
    count(level = .data[[moderator]], name = "effect_sizes") |>
    mutate(
      studies = vapply(
        level,
        function(x) n_distinct(data$study_id[usable & data[[moderator]] == x]),
        integer(1)
      ),
      include = effect_sizes >= min_k
    )

  # Retain levels represented by at least five effect sizes.
  keep_levels <- level_counts |>
    filter(include) |>
    pull(level)

  # Use the same row positions to subset the data and VCV matrix.
  idx <- which(usable & data[[moderator]] %in% keep_levels)
  dat <- data[idx, , drop = FALSE]
  V_subset <- V[idx, idx, drop = FALSE]

  # Remove unused factor levels after filtering.
  dat[[moderator]] <- droplevels(factor(dat[[moderator]]))

  # A categorical moderator requires at least two retained levels.
  if (nlevels(dat[[moderator]]) < 2) {
    stop("Fewer than two levels remain for ", moderator, ".")
  }

  list(data = dat, V = V_subset, level_counts = level_counts)
}

prepare_continuous <- function(data, V, moderator) {
  # Retain rows with finite moderator values.
  idx <- which(is.finite(data[[moderator]]))

  # Subset the data and VCV matrix with the same row positions.
  dat <- data[idx, , drop = FALSE]
  V_subset <- V[idx, idx, drop = FALSE]

  # Store the moderator as a numeric variable.
  dat[[moderator]] <- as.numeric(dat[[moderator]])

  # A continuous moderator requires at least two distinct values.
  if (n_distinct(dat[[moderator]]) < 2) {
    stop("The moderator does not contain at least two values: ", moderator, ".")
  }

  list(data = dat, V = V_subset, level_counts = NULL)
}

Fit one uni-moderator model

The function below prepares one moderator, subsets the complete VCV matrix, and fits its model. For categorical moderators with at least three retained levels, it also calculates unadjusted Tukey contrasts with a separate no-intercept model.

Code 
fit_unimoderator <- function(
  data,
  V,
  yi,
  effect_label,
  effect_axis_label,
  moderator,
  label,
  kind,
  missing_label = NA_character_,
  min_k = 5
) {
  # Apply the appropriate preparation rule for the moderator type.
  prepared <- if (kind == "categorical") {
    prepare_categorical(data, V, moderator, missing_label, min_k)
  } else {
    prepare_continuous(data, V, moderator)
  }

  model_data <- prepared$data
  V_subset <- prepared$V

  # Fit one multilevel meta-regression with the subsetted VCV matrix.
  model <- metafor::rma.mv(
    yi = model_data[[yi]],
    V = V_subset,
    mods = reformulate(moderator),
    random = list(
      ~ 1 | study_id,
      ~ 1 | es_id,
      ~ 1 | strain
    ),
    test = "t",
    method = "REML",
    sparse = TRUE,
    data = model_data
  )

  # Draw an orchard plot for a categorical moderator. We draw continuous
  # moderators with metafor::regplot() when we display the model results.
  model_plot <- if (kind == "categorical") {
    orchaRd::orchard_plot(
      model,
      mod = moderator,
      group = "study_id",
      xlab = effect_axis_label,
      flip = TRUE,
      angle = 0,
      legend.pos = "top.left"
    ) +
      # Use the complete named palette so shared categories keep their colour.
      ggplot2::scale_colour_manual(
        values = category_palettes[[moderator]],
        drop = FALSE
      ) +
      ggplot2::scale_fill_manual(
        values = category_palettes[[moderator]],
        drop = FALSE
      )
  } else {
    NULL
  }

  contrasts <- NULL

  # Calculate unadjusted Tukey contrasts when at least three levels remain.
  if (kind == "categorical" && nlevels(model_data[[moderator]]) >= 3) {
    # Fit a no-intercept model so that each coefficient represents one level.
    no_intercept_model <- metafor::rma.mv(
      yi = model_data[[yi]],
      V = V_subset,
      mods = reformulate(moderator, intercept = FALSE),
      random = list(
        ~ 1 | study_id,
        ~ 1 | es_id,
        ~ 1 | strain
      ),
      test = "t",
      method = "REML",
      sparse = TRUE,
      data = model_data
    )

    # Create all pairwise Tukey contrasts among the retained levels.
    contrast_matrix <- multcomp::contrMat(
      table(model_data[[moderator]]),
      type = "Tukey"
    )
    colnames(contrast_matrix) <- names(coef(no_intercept_model))

    # Report the unadjusted p-value for each contrast.
    contrasts <- summary(
      multcomp::glht(no_intercept_model, linfct = contrast_matrix),
      test = multcomp::adjusted("none")
    )
  }

  list(
    effect = yi,
    effect_label = effect_label,
    effect_axis_label = effect_axis_label,
    variable = moderator,
    label = label,
    kind = kind,
    data = model_data,
    V = V_subset,
    level_counts = prepared$level_counts,
    model = model,
    r2 = orchaRd::r2_ml(model),
    plot = model_plot,
    contrasts = contrasts
  )
}

Fit all uni-moderator models

We call fit_unimoderator() once for each combination of effect-size type and moderator. The function subsets the corresponding complete VCV matrix to match the effect sizes retained for each model.

# Fit every moderator separately for lnRR, lnVR, and lnCVR.
unimoderator_results <- purrr::map(
  effect_specs$effect,
  function(effect) {
    effect_label <- effect_specs$label[effect_specs$effect == effect]
    effect_axis_label <- effect_specs$axis_label[effect_specs$effect == effect]
    input <- metric_inputs[[effect]]

    results <- purrr::pmap(
      moderators |>
        dplyr::select(variable, label, kind, missing_label),
      function(variable, label, kind, missing_label) {
        fit_unimoderator(
          data = input$data,
          V = input$V,
          yi = effect,
          effect_label = effect_label,
          effect_axis_label = effect_axis_label,
          moderator = variable,
          label = label,
          kind = kind,
          missing_label = missing_label,
          min_k = 5
        )
      }
    )

    names(results) <- moderators$variable
    results
  }
)

title: "Uni-moderators"

We fit one uni-moderator multilevel meta-regression for each study characteristic and effect-size type. The models use lnRR, lnVR, or lnCVR as the response and the same random-effects structure as the overall-effect model: study (`study_id`), effect size (`es_id`), and strain (`strain`). We construct a complete VCV matrix for each effect-size type with `rho = 0.5`. For categorical moderators, we remove levels with fewer than five effect sizes ($k < 5$) and subset the corresponding rows and columns of the matching VCV matrix.

::: callout-note
## Variable definitions

| Variable | Definition |
|:--|:--|
| $k$ | Number of effect sizes in a model. |
| `lnRR` | Log response ratio, which measures differences in group means. |
| `lnVR` | Log variability ratio, which measures absolute variability. |
| `lnCVR` | Log coefficient of variation ratio, which measures variability relative to the group mean. |
| `V` | Sampling variance-covariance matrix for one effect-size type. |
| $\rho$ | Working correlation for multiple or repeated outcomes that share a group. |
| `QM` | Omnibus test statistic for the moderator. |
| Marginal $R^2$ | Proportion of model variance associated with the moderator. |
| Conditional $R^2$ | Proportion of model variance associated with the moderator and random effects. |
:::

Across every lnRR plot, negative values represent less anxiety- or depression-like behaviour and positive values represent more anxiety- or depression-like behaviour. For lnVR and lnCVR, negative values represent lower variability and positive values represent greater variability in the treatment group.

### Prepare the moderator data

For a categorical moderator, we remove levels with $k < 5$ and subset the same rows and columns from `V`. For a continuous moderator, we retain finite values and subset `V` in the same way.

```{r}
#| label: prepare-moderator-data
#| code-fold: true

prepare_categorical <- function(data, V, moderator, missing_label = NA_character_, min_k = 5) {
  # Start with rows that contain a recorded moderator value.
  usable <- !is.na(data[[moderator]])

  # Treat a specified text label as missing when required.
  if (!is.na(missing_label)) {
    usable <- usable & as.character(data[[moderator]]) != missing_label
  }

  # Count the effect sizes in each moderator level.
  level_counts <- data[usable, ] |>
    count(level = .data[[moderator]], name = "effect_sizes") |>
    mutate(
      studies = vapply(
        level,
        function(x) n_distinct(data$study_id[usable & data[[moderator]] == x]),
        integer(1)
      ),
      include = effect_sizes >= min_k
    )

  # Retain levels represented by at least five effect sizes.
  keep_levels <- level_counts |>
    filter(include) |>
    pull(level)

  # Use the same row positions to subset the data and VCV matrix.
  idx <- which(usable & data[[moderator]] %in% keep_levels)
  dat <- data[idx, , drop = FALSE]
  V_subset <- V[idx, idx, drop = FALSE]

  # Remove unused factor levels after filtering.
  dat[[moderator]] <- droplevels(factor(dat[[moderator]]))

  # A categorical moderator requires at least two retained levels.
  if (nlevels(dat[[moderator]]) < 2) {
    stop("Fewer than two levels remain for ", moderator, ".")
  }

  list(data = dat, V = V_subset, level_counts = level_counts)
}

prepare_continuous <- function(data, V, moderator) {
  # Retain rows with finite moderator values.
  idx <- which(is.finite(data[[moderator]]))

  # Subset the data and VCV matrix with the same row positions.
  dat <- data[idx, , drop = FALSE]
  V_subset <- V[idx, idx, drop = FALSE]

  # Store the moderator as a numeric variable.
  dat[[moderator]] <- as.numeric(dat[[moderator]])

  # A continuous moderator requires at least two distinct values.
  if (n_distinct(dat[[moderator]]) < 2) {
    stop("The moderator does not contain at least two values: ", moderator, ".")
  }

  list(data = dat, V = V_subset, level_counts = NULL)
}
```

### Fit one uni-moderator model

The function below prepares one moderator, subsets the complete VCV matrix, and fits its model. For categorical moderators with at least three retained levels, it also calculates unadjusted Tukey contrasts with a separate no-intercept model.

```{r}
#| label: fit-one-unimoderator
#| code-fold: true

fit_unimoderator <- function(
  data,
  V,
  yi,
  effect_label,
  effect_axis_label,
  moderator,
  label,
  kind,
  missing_label = NA_character_,
  min_k = 5
) {
  # Apply the appropriate preparation rule for the moderator type.
  prepared <- if (kind == "categorical") {
    prepare_categorical(data, V, moderator, missing_label, min_k)
  } else {
    prepare_continuous(data, V, moderator)
  }

  model_data <- prepared$data
  V_subset <- prepared$V

  # Fit one multilevel meta-regression with the subsetted VCV matrix.
  model <- metafor::rma.mv(
    yi = model_data[[yi]],
    V = V_subset,
    mods = reformulate(moderator),
    random = list(
      ~ 1 | study_id,
      ~ 1 | es_id,
      ~ 1 | strain
    ),
    test = "t",
    method = "REML",
    sparse = TRUE,
    data = model_data
  )

  # Draw an orchard plot for a categorical moderator. We draw continuous
  # moderators with metafor::regplot() when we display the model results.
  model_plot <- if (kind == "categorical") {
    orchaRd::orchard_plot(
      model,
      mod = moderator,
      group = "study_id",
      xlab = effect_axis_label,
      flip = TRUE,
      angle = 0,
      legend.pos = "top.left"
    ) +
      # Use the complete named palette so shared categories keep their colour.
      ggplot2::scale_colour_manual(
        values = category_palettes[[moderator]],
        drop = FALSE
      ) +
      ggplot2::scale_fill_manual(
        values = category_palettes[[moderator]],
        drop = FALSE
      )
  } else {
    NULL
  }

  contrasts <- NULL

  # Calculate unadjusted Tukey contrasts when at least three levels remain.
  if (kind == "categorical" && nlevels(model_data[[moderator]]) >= 3) {
    # Fit a no-intercept model so that each coefficient represents one level.
    no_intercept_model <- metafor::rma.mv(
      yi = model_data[[yi]],
      V = V_subset,
      mods = reformulate(moderator, intercept = FALSE),
      random = list(
        ~ 1 | study_id,
        ~ 1 | es_id,
        ~ 1 | strain
      ),
      test = "t",
      method = "REML",
      sparse = TRUE,
      data = model_data
    )

    # Create all pairwise Tukey contrasts among the retained levels.
    contrast_matrix <- multcomp::contrMat(
      table(model_data[[moderator]]),
      type = "Tukey"
    )
    colnames(contrast_matrix) <- names(coef(no_intercept_model))

    # Report the unadjusted p-value for each contrast.
    contrasts <- summary(
      multcomp::glht(no_intercept_model, linfct = contrast_matrix),
      test = multcomp::adjusted("none")
    )
  }

  list(
    effect = yi,
    effect_label = effect_label,
    effect_axis_label = effect_axis_label,
    variable = moderator,
    label = label,
    kind = kind,
    data = model_data,
    V = V_subset,
    level_counts = prepared$level_counts,
    model = model,
    r2 = orchaRd::r2_ml(model),
    plot = model_plot,
    contrasts = contrasts
  )
}
```
