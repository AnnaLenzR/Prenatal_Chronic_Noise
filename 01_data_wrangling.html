Derivation of missing summary statistics

Calculation methods

  • We estimate a missing mean from Q1, the median, Q3, and sample size with the Luo et al. method (Luo et al. 2018).
  • We estimate a missing SD from Q1, Q3, and sample size with the Wan et al. method (Wan et al. 2014).
  • When the source reports an SE but not an SD, we calculate SD = SE × sqrt(n).
  • When the source does not report an SE, we calculate SE = SD / sqrt(n).
  • We retain existing extracted values unchanged.

\[ \widehat{M} = \left(0.7 + \frac{0.39}{n}\right)\frac{Q_1 + Q_3}{2} + \left(0.3 - \frac{0.39}{n}\right)M_d \]

\[ \widehat{SD} = \frac{Q_3-Q_1} {2\Phi^{-1}\left(\frac{0.75n-0.125}{n+0.25}\right)} \]

We define one function for each published estimator and one function that completes the mean, SD, and SE for either treatment arm. The function also records the source of every analysis value.

estimate_mean_luo <- function(q1, median, q3, n) {
  weight <- 0.7 + 0.39 / n
  weight * ((q1 + q3) / 2) + (1 - weight) * median
}

estimate_sd_wan <- function(q1, q3, n) {
  denominator <- 2 * stats::qnorm((0.75 * n - 0.125) / (n + 0.25))
  (q3 - q1) / denominator
}

complete_arm_statistics <- function(data, arm) {
  n <- data[[paste0(arm, "_n")]]
  mean_source <- data[[paste0(arm, "_mean")]]
  sd_source <- data[[paste0(arm, "_sd")]]
  se_source <- data[[paste0(arm, "_se")]]
  median <- data[[paste0(arm, "_median")]]
  q1 <- data[[paste0(arm, "_q1")]]
  q3 <- data[[paste0(arm, "_q3")]]

  mean_from_iqr <- is.na(mean_source) & complete.cases(q1, median, q3, n)
  sd_from_iqr <- is.na(sd_source) & complete.cases(q1, q3, n)
  sd_from_se <- is.na(sd_source) & !sd_from_iqr & !is.na(se_source) & !is.na(n)

  if (any(mean_from_iqr & !(q1 <= median & median <= q3))) {
    stop("Invalid Q1 <= median <= Q3 ordering in the ", arm, " arm.")
  }

  mean_analysis <- mean_source
  mean_analysis[mean_from_iqr] <- estimate_mean_luo(
    q1[mean_from_iqr], median[mean_from_iqr], q3[mean_from_iqr], n[mean_from_iqr]
  )

  sd_analysis <- sd_source
  sd_analysis[sd_from_iqr] <- estimate_sd_wan(
    q1[sd_from_iqr], q3[sd_from_iqr], n[sd_from_iqr]
  )
  sd_analysis[sd_from_se] <- se_source[sd_from_se] * sqrt(n[sd_from_se])

  se_analysis <- se_source
  se_from_sd <- is.na(se_analysis) & !is.na(sd_analysis) & !is.na(n)
  se_analysis[se_from_sd] <- sd_analysis[se_from_sd] / sqrt(n[se_from_sd])

  data[[paste0(arm, "_mean_analysis")]] <- mean_analysis
  data[[paste0(arm, "_mean_method")]] <- if_else(
    mean_from_iqr,
    "Luo et al. (2018): Q1, median, Q3, and n",
    "Extracted mean used unchanged"
  )
  data[[paste0(arm, "_sd_analysis")]] <- sd_analysis
  data[[paste0(arm, "_sd_method")]] <- case_when(
    sd_from_iqr ~ "Wan et al. (2014): Q1, Q3, and n",
    sd_from_se ~ "SD = SE * sqrt(n)",
    TRUE ~ "Extracted SD used unchanged"
  )
  data[[paste0(arm, "_se_analysis")]] <- se_analysis
  data[[paste0(arm, "_se_method")]] <- if_else(
    se_from_sd,
    "SE = SD / sqrt(n)",
    "Extracted SE used unchanged"
  )

  if (anyNA(mean_analysis) || anyNA(sd_analysis) || anyNA(se_analysis)) {
    stop("Unresolved summary statistic in the ", arm, " arm.")
  }

  data
}

## Data checks

We check the identifiers, summary statistics, sample sizes, and coded variables before exporting the analysis-ready database. The chapter stops if any check fails.

```{r}
#| label: validate-analysis-data

validation <- tibble(
  check = c(
    "One row per es_id",
    "Study identifiers present",
    "Positive sample sizes",
    "Complete means",
    "Complete non-negative SDs",
    "Complete non-negative SEs",
    "Valid higher_better values",
    "Valid comparison_structure values"
  ),
  failures = c(
    sum(is.na(analysis_ready$es_id)) + sum(duplicated(analysis_ready$es_id)),
    sum(is.na(analysis_ready$study_id) | analysis_ready$study_id == ""),
    sum(is.na(analysis_ready$c_n) | is.na(analysis_ready$ex_n) |
          analysis_ready$c_n <= 0 | analysis_ready$ex_n <= 0),
    sum(is.na(analysis_ready$c_mean) | is.na(analysis_ready$ex_mean)),
    sum(is.na(analysis_ready$c_sd) | is.na(analysis_ready$ex_sd) |
          analysis_ready$c_sd < 0 | analysis_ready$ex_sd < 0),
    sum(is.na(analysis_ready$c_se) | is.na(analysis_ready$ex_se) |
          analysis_ready$c_se < 0 | analysis_ready$ex_se < 0),
    sum(!analysis_ready$higher_better %in% c("yes", "no")),
    sum(!analysis_ready$comparison_structure %in% c("independent", "dependent"))
  )
) |>
  mutate(status = if_else(failures == 0, "Pass", "Fail"))

validation |>
  knitr::kable(caption = "Analysis-ready data checks.")

if (any(validation$status == "Fail")) {
  stop("The analysis-ready database failed validation.")
}
```
