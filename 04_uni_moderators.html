5.1 Approach

Each candidate moderator from the protocol (biological factors: species, sex, age; methodological factors: exposure timing/intensity, noise type, outcome type, assay type) is fit one at a time (“uni-moderator”), using the same random-effects structure and cohort-level variance-covariance matrix as the overall-effect model in Chapter 3. Fitting moderators one at a time, rather than in one large multi-moderator model, matches the protocol and keeps every model interpretable given the sample size (207 effect sizes from 16 studies).

For categorical moderators, a level is only modelled if it has at least 5 effect sizes; levels below that are dropped and reported as such, rather than silently included in an unstable estimate. The number of studies contributing to each level is always shown next to the number of effect sizes, because several effect sizes from one study are weaker evidence than the same count spread across studies. For categorical moderators with 3 or more surviving levels, pairwise contrasts between levels are also reported, unadjusted for multiple comparisons (as descriptive contrasts, not confirmatory tests). For continuous moderators, the model is a simple linear meta-regression.

offspring_age_class (juvenile / adolescent / adult) is intentionally not in this main set - per the protocol deviation addendum, it is restricted to descriptive summaries and sensitivity analyses and does not replace offspring_age_d as the primary age moderator. It is fit at the end of this chapter, labelled as a sensitivity analysis, not a primary result.

# Keep only levels with at least min_k effect sizes. The VCV matrix does not
# need to be subset here because it is rebuilt from the filtered data (see
# fit_categorical()) rather than sliced from the full matrix.
filter_low_k <- function(dat, moderator, min_k = 5) {
  dat <- dat %>% filter(!is.na(.data[[moderator]]))
  dat[[moderator]] <- droplevels(as.factor(dat[[moderator]]))

  level_counts <- dat %>%
    group_by(level = .data[[moderator]]) %>%
    summarise(n_effect_sizes = n(), n_studies = n_distinct(study_id), .groups = "drop")

  keep_levels <- level_counts %>% filter(n_effect_sizes >= min_k) %>% pull(level)
  dat_f <- dat %>% filter(.data[[moderator]] %in% keep_levels)
  dat_f[[moderator]] <- droplevels(dat_f[[moderator]])

  list(data = dat_f, level_counts = level_counts)
}

fit_categorical <- function(es, moderator, min_k = 5) {
  filtered <- filter_low_k(es, moderator, min_k)
  dat <- filtered$data
  if (nrow(dat) < 5 || nlevels(dat[[moderator]]) < 2) return(list(level_counts = filtered$level_counts, model = NULL))

  V <- metafor::vcalc(vi = lnrr_var, cluster = dependency_cluster_id, obs = es_id, rho = 0.5, data = dat)

  mod <- metafor::rma.mv(
    yi = lnrr_harm,
    V = V,
    mods = as.formula(paste0("~ ", moderator)),
    random = list(~1 | study_id, ~1 | es_id, ~1 | strain),
    test = "t",
    method = "REML",
    sparse = TRUE,
    data = dat
  )

  plot <- orchaRd::orchard_plot(
    mod,
    mod = moderator,
    group = "study_id",
    xlab = "lnRR (harmonized)",
    flip = TRUE
  )

  tukey <- NULL
  if (nlevels(dat[[moderator]]) >= 3) {
    mod_nointercept <- metafor::rma.mv(
      yi = lnrr_harm,
      V = V,
      mods = as.formula(paste0("~ ", moderator, " - 1")),
      random = list(~1 | study_id, ~1 | es_id, ~1 | strain),
      test = "t",
      method = "REML",
      sparse = TRUE,
      data = dat
    )
    k <- nlevels(dat[[moderator]])
    K <- multcomp::contrMat(rep(1, k), type = "Tukey")
    colnames(K) <- names(coef(mod_nointercept))
    tukey <- summary(multcomp::glht(mod_nointercept, linfct = K), test = multcomp::adjusted("none"))
  }

  list(level_counts = filtered$level_counts, model = mod, r2 = orchaRd::r2_ml(mod), plot = plot, tukey = tukey)
}

fit_continuous <- function(es, moderator, min_n = 10, min_studies = 3) {
  dat <- es %>% filter(!is.na(.data[[moderator]]))
  if (nrow(dat) < min_n || n_distinct(dat$study_id) < min_studies) {
    return(list(n_effect_sizes = nrow(dat), n_studies = n_distinct(dat$study_id), model = NULL))
  }

  V <- metafor::vcalc(vi = lnrr_var, cluster = dependency_cluster_id, obs = es_id, rho = 0.5, data = dat)

  mod <- metafor::rma.mv(
    yi = lnrr_harm,
    V = V,
    mods = as.formula(paste0("~ ", moderator)),
    random = list(~1 | study_id, ~1 | es_id, ~1 | strain),
    test = "t",
    method = "REML",
    sparse = TRUE,
    data = dat
  )

  plot <- orchaRd::bubble_plot(
    mod,
    mod = moderator,
    group = "study_id",
    xlab = moderator,
    ylab = "lnRR (harmonized)"
  )

  list(n_effect_sizes = nrow(dat), n_studies = n_distinct(dat$study_id), model = mod, r2 = orchaRd::r2_ml(mod), plot = plot)
}

## Approach

Each candidate moderator from the protocol (biological factors: species, sex, age; methodological factors: exposure timing/intensity, noise type, outcome type, assay type) is fit **one at a time** ("uni-moderator"), using the same random-effects structure and cohort-level variance-covariance matrix as the overall-effect model in Chapter 3. Fitting moderators one at a time, rather than in one large multi-moderator model, matches the protocol and keeps every model interpretable given the sample size (207 effect sizes from 16 studies).

For categorical moderators, a level is only modelled if it has at least 5 effect sizes; levels below that are dropped and reported as such, rather than silently included in an unstable estimate. The number of *studies* contributing to each level is always shown next to the number of effect sizes, because several effect sizes from one study are weaker evidence than the same count spread across studies. For categorical moderators with 3 or more surviving levels, pairwise contrasts between levels are also reported, unadjusted for multiple comparisons (as descriptive contrasts, not confirmatory tests). For continuous moderators, the model is a simple linear meta-regression.

`offspring_age_class` (juvenile / adolescent / adult) is intentionally not in this main set - per the protocol deviation addendum, it is restricted to descriptive summaries and sensitivity analyses and does not replace `offspring_age_d` as the primary age moderator. It is fit at the end of this chapter, labelled as a sensitivity analysis, not a primary result.

# Keep only levels with at least min_k effect sizes. The VCV matrix does not
# need to be subset here because it is rebuilt from the filtered data (see
# fit_categorical()) rather than sliced from the full matrix.
filter_low_k <- function(dat, moderator, min_k = 5) {
  dat <- dat %>% filter(!is.na(.data[[moderator]]))
  dat[[moderator]] <- droplevels(as.factor(dat[[moderator]]))

  level_counts <- dat %>%
    group_by(level = .data[[moderator]]) %>%
    summarise(n_effect_sizes = n(), n_studies = n_distinct(study_id), .groups = "drop")

  keep_levels <- level_counts %>% filter(n_effect_sizes >= min_k) %>% pull(level)
  dat_f <- dat %>% filter(.data[[moderator]] %in% keep_levels)
  dat_f[[moderator]] <- droplevels(dat_f[[moderator]])

  list(data = dat_f, level_counts = level_counts)
}

fit_categorical <- function(es, moderator, min_k = 5) {
  filtered <- filter_low_k(es, moderator, min_k)
  dat <- filtered$data
  if (nrow(dat) < 5 || nlevels(dat[[moderator]]) < 2) return(list(level_counts = filtered$level_counts, model = NULL))

  V <- metafor::vcalc(vi = lnrr_var, cluster = dependency_cluster_id, obs = es_id, rho = 0.5, data = dat)

  mod <- metafor::rma.mv(
    yi = lnrr_harm,
    V = V,
    mods = as.formula(paste0("~ ", moderator)),
    random = list(~1 | study_id, ~1 | es_id, ~1 | strain),
    test = "t",
    method = "REML",
    sparse = TRUE,
    data = dat
  )

  plot <- orchaRd::orchard_plot(
    mod,
    mod = moderator,
    group = "study_id",
    xlab = "lnRR (harmonized)",
    flip = TRUE
  )

  tukey <- NULL
  if (nlevels(dat[[moderator]]) >= 3) {
    mod_nointercept <- metafor::rma.mv(
      yi = lnrr_harm,
      V = V,
      mods = as.formula(paste0("~ ", moderator, " - 1")),
      random = list(~1 | study_id, ~1 | es_id, ~1 | strain),
      test = "t",
      method = "REML",
      sparse = TRUE,
      data = dat
    )
    k <- nlevels(dat[[moderator]])
    K <- multcomp::contrMat(rep(1, k), type = "Tukey")
    colnames(K) <- names(coef(mod_nointercept))
    tukey <- summary(multcomp::glht(mod_nointercept, linfct = K), test = multcomp::adjusted("none"))
  }

  list(level_counts = filtered$level_counts, model = mod, r2 = orchaRd::r2_ml(mod), plot = plot, tukey = tukey)
}

fit_continuous <- function(es, moderator, min_n = 10, min_studies = 3) {
  dat <- es %>% filter(!is.na(.data[[moderator]]))
  if (nrow(dat) < min_n || n_distinct(dat$study_id) < min_studies) {
    return(list(n_effect_sizes = nrow(dat), n_studies = n_distinct(dat$study_id), model = NULL))
  }

  V <- metafor::vcalc(vi = lnrr_var, cluster = dependency_cluster_id, obs = es_id, rho = 0.5, data = dat)

  mod <- metafor::rma.mv(
    yi = lnrr_harm,
    V = V,
    mods = as.formula(paste0("~ ", moderator)),
    random = list(~1 | study_id, ~1 | es_id, ~1 | strain),
    test = "t",
    method = "REML",
    sparse = TRUE,
    data = dat
  )

  plot <- orchaRd::bubble_plot(
    mod,
    mod = moderator,
    group = "study_id",
    xlab = moderator,
    ylab = "lnRR (harmonized)"
  )

  list(n_effect_sizes = nrow(dat), n_studies = n_distinct(dat$study_id), model = mod, r2 = orchaRd::r2_ml(mod), plot = plot)
}
