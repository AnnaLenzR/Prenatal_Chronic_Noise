7  CRIME-Q critical appraisal

This chapter reports the adapted CRIME-Q critical-appraisal items used in this review and the final, reconciled study-level ratings. The assessment covers quality of reporting (QoR), methodological quality (MQ), and risk of bias (RoB), together with three review-specific transparency items, as described in the protocol. Inter-reviewer agreement (Cohen’s kappa) is not reported here.

read_appraisal <- function(path) {
  read_excel(
    path = path,
    sheet = input_sheet,
    col_types = "text",
    na = ""
  ) |>
    rename_with(str_trim) |>
    mutate(across(everything(), ~ if_else(is.na(.x), NA_character_, str_trim(.x)))) |>
    # A stray ".pdf" suffix on one study_id in this workbook doesn't match the
    # study_id used everywhere else in the project; fix it here on read.
    mutate(study_id = str_remove(study_id, "\\.pdf$"))
}

validate_appraisal <- function(dat, source_name) {
  required_cols <- c("study_id", "study_title", item_spec$data_column)
  missing_cols <- setdiff(required_cols, names(dat))

  if (length(missing_cols) > 0L) {
    stop(source_name, " is missing required column(s): ", paste(missing_cols, collapse = ", "))
  }
  if (anyDuplicated(dat$study_id) > 0L) {
    duplicated_ids <- unique(dat$study_id[duplicated(dat$study_id)])
    stop(source_name, " contains duplicated study_id value(s): ", paste(duplicated_ids, collapse = ", "))
  }

  rating_dat <- dat |>
    select(study_id, all_of(item_spec$data_column)) |>
    pivot_longer(cols = all_of(item_spec$data_column), names_to = "data_column", values_to = "Rating")

  if (anyNA(rating_dat$Rating)) {
    missing_ratings <- rating_dat |>
      filter(is.na(Rating)) |>
      transmute(location = paste(study_id, data_column, sep = " / ")) |>
      pull(location)
    stop(source_name, " contains missing CRIME-Q rating(s): ", paste(missing_ratings, collapse = "; "))
  }

  unexpected <- setdiff(unique(rating_dat$Rating), rating_levels)
  if (length(unexpected) > 0L) {
    stop(source_name, " contains unexpected rating value(s): ", paste(unexpected, collapse = ", "))
  }

  invisible(dat)
}

study_label <- function(study_id) {
  parts <- str_split(study_id, "_", simplify = FALSE)
  vapply(
    parts,
    function(x) {
      if (length(x) < 2L) return(x[1])
      paste0(str_to_title(x[1]), " et al. ", x[2])
    },
    character(1)
  )
}
